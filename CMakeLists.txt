cmake_minimum_required(VERSION 3.24)
project(TrafficAssignmentApproaches VERSION 0.1.0 LANGUAGES CXX)

if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
endif()

option(BUILD_PYTHON_BINDINGS "Build traffic_assignment._core" OFF)
if(SKBUILD)
  set(TA_APPS_DEFAULT OFF)
else()
  set(TA_APPS_DEFAULT ON)
endif()
option(BUILD_CPP_APPS "Build standalone TAP and CNDP applications" ${TA_APPS_DEFAULT})
option(BUILD_NATIVE_TESTS "Build tests of the public C++ API" OFF)

include(cmake/Dependencies.cmake)
find_package(Threads REQUIRED)
if(BUILD_NATIVE_TESTS)
  enable_testing()
endif()
add_subdirectory(cpp)
